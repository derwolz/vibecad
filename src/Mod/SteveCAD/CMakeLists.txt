# Native cmake --build does not go through rattler create_bundle.sh, so pin
# Codex here or the module keeps a leftover runtime from an older tree.
option(STEVECAD_INSTALL_CODEX_RUNTIME "Install the pinned Codex app-server into the build module" ON)
if(UNIX AND STEVECAD_INSTALL_CODEX_RUNTIME)
    if(NOT Python3_EXECUTABLE)
        message(FATAL_ERROR "Python3_EXECUTABLE is required to install the Codex app-server runtime")
    endif()
    set(STEVECAD_CODEX_INSTALLER
        "${CMAKE_SOURCE_DIR}/package/rattler-build/scripts/install_stevecad_codex_runtime.sh")
    set(STEVECAD_CODEX_MODULE_DIR "${CMAKE_BINARY_DIR}/Mod/SteveCAD")
    set(STEVECAD_CODEX_RUNTIME_STAMP "${STEVECAD_CODEX_MODULE_DIR}/codex_runtime/runtime.json")
    # The installer validates its cache cheaply. A manifest timestamp alone
    # cannot detect removed companion binaries or stale runtime metadata.
    add_custom_target(SteveCADCodexRuntime ALL
        BYPRODUCTS "${STEVECAD_CODEX_RUNTIME_STAMP}"
        COMMAND "${CMAKE_COMMAND}" -E make_directory "${STEVECAD_CODEX_MODULE_DIR}"
        COMMAND
            bash "${STEVECAD_CODEX_INSTALLER}"
            "${Python3_EXECUTABLE}"
            "${STEVECAD_CODEX_MODULE_DIR}"
        DEPENDS
            "${STEVECAD_CODEX_INSTALLER}"
            "${CMAKE_CURRENT_SOURCE_DIR}/SteveCADCodex.py"
        COMMENT "Verifying pinned Codex app-server in ${STEVECAD_CODEX_MODULE_DIR}"
        VERBATIM
    )
    add_dependencies(SteveCADScripts SteveCADCodexRuntime)
    install(
        DIRECTORY "${STEVECAD_CODEX_MODULE_DIR}/codex_runtime"
        DESTINATION Mod/SteveCAD
        USE_SOURCE_PERMISSIONS
        OPTIONAL
    )
endif()
